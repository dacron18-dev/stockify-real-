cmake_minimum_required(VERSION 3.10)
project(stockify CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
include_directories(third_party src)
add_executable(stockify src/main.cpp)
target_compile_definitions(stockify PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
target_link_libraries(stockify OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
add_executable(concepts_demo demo/concepts_demo.cpp)
target_link_libraries(concepts_demo Threads::Threads)
